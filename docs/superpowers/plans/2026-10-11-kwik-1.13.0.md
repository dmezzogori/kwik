- [ ] **Step 1: Make the API test client transactional.** Today `client` overrides `_get_session` with a plain session that is never committed. Once kwik commits on every route, that would write real rows that leak across tests. Join the session into an outer transaction so `commit()` only releases a SAVEPOINT:

```python
@pytest.fixture
def client(kwik_app: Kwik, engine: Engine) -> Generator[TestClient, None, None]:
    """
    Set up an HTTP client whose requests share one rolled-back transaction.

    The session joins an outer transaction in ``create_savepoint`` mode, so the
    commit kwik issues before every response only releases a SAVEPOINT and the
    test's writes are discarded at teardown.
    """
    connection = engine.connect()
    transaction = connection.begin()
    api_session = Session(bind=connection, expire_on_commit=False, join_transaction_mode="create_savepoint")

    def _get_session_override() -> Generator[Session, None, None]:
        yield api_session

    kwik_app.app.dependency_overrides[_get_session] = _get_session_override
    try:
        with TestClient(app=kwik_app.app) as client:
            yield client
    finally:
        kwik_app.app.dependency_overrides.pop(_get_session, None)
        api_session.close()
        transaction.rollback()
        connection.close()
```

```python
from __future__ import annotations

from http import HTTPStatus
from typing import TYPE_CHECKING
from unittest.mock import MagicMock

from fastapi import APIRouter, HTTPException
from fastapi.responses import StreamingResponse
from fastapi.testclient import TestClient
from sqlalchemy import select

from kwik.applications import Kwik
from kwik.database import create_session
from kwik.dependencies import Session
from kwik.dependencies.session import _get_session
from kwik.models import Role

if TYPE_CHECKING:
    from collections.abc import Iterator

    from sqlalchemy import Engine

    from kwik.settings import BaseKwikSettings

ROLE_NAME = "commit-before-response-probe"


def _recording(app, events: list[str], on_start=None):
    """Wrap an ASGI app, appending "response.start" when the status line goes out."""

    async def wrapped(scope, receive, send):
        async def recording_send(message):
            if message["type"] == "http.response.start":
                if on_start is not None:
                    on_start()
                events.append("response.start")
            await send(message)

        await app(scope, receive, recording_send)

    return wrapped
```

```python
def _commit_session(session: Annotated[_Session, Depends(_get_session)]) -> Generator[None, None, None]:
    """
    Commit the request's session once the endpoint returns, before the response is sent.

    Installed app-wide by ``Kwik`` with ``scope="function"``: FastAPI closes
    function-scoped dependencies after the endpoint ran and its response was
    serialized, but before the status line goes out, so a client that acts on
    the response at once finds the write committed. It shares the request's
    cached session; ``_get_session`` still owns the session, rolls back on
    error, commits writes made later (streaming bodies, background tasks)
    and closes it after the response.

    If the endpoint raised, the exception passes through this generator before
    the commit line and nothing is committed.
    """
    yield
    session.commit()
```

In `kwik.py`, import `Depends` from fastapi and `_commit_session`, and pass `dependencies=[Depends(_commit_session, scope="function")]` to `FastAPI(...)`. Add a sentence to the `Kwik` class docstring.

```python
GRACEFUL_SHUTDOWN_MARGIN = 5
"""Seconds left between uvicorn's shutdown deadline and gunicorn's SIGKILL.

Covers the cleanup that cancelled tasks run (e.g. shielded lease releases) and the
lifespan shutdown."""


class KwikUvicornWorker(UvicornWorker):
    """Uvicorn worker with optimized configuration for HTTP tools and WebSocket support."""

    CONFIG_KWARGS: ClassVar[dict[str, str | bool]] = {"http": "httptools", "ws": "websockets", "proxy_headers": True}

    def __init__(self, *args: Any, **kwargs: Any) -> None:  # noqa: ANN401
        """
        Build the worker and bound uvicorn's graceful shutdown by gunicorn's.

        Without a bound, uvicorn waits on open streaming responses until gunicorn
        SIGKILLs the worker after ``graceful_timeout``, and the streams' cleanup
        never runs. Deriving the value from gunicorn's keeps the two in step.
        """
        super().__init__(*args, **kwargs)
        self.config.timeout_graceful_shutdown = max(int(self.cfg.graceful_timeout) - GRACEFUL_SHUTDOWN_MARGIN, 1)
```

```python
def warn_if_pool_exceeds_connections(settings: BaseKwikSettings) -> bool:
    """
    Log a warning when the worker pools together can exceed the database's connection limit.

    Each worker process has its own pool of ``POSTGRES_POOL_SIZE + POSTGRES_MAX_OVERFLOW``
    connections. Returns True when it warned.
    """
    per_worker = settings.POSTGRES_POOL_SIZE + settings.POSTGRES_MAX_OVERFLOW
    total = settings.BACKEND_WORKERS * per_worker
    if total <= settings.POSTGRES_MAX_CONNECTIONS:
        return False
    logger.warning(
        f"{settings.BACKEND_WORKERS} workers x {per_worker} pooled connections = {total} "
        f"exceeds POSTGRES_MAX_CONNECTIONS={settings.POSTGRES_MAX_CONNECTIONS}",
    )
    return True
```

```python
def build_api_router(*, exclude: Iterable[tuple[str, str]] = ()) -> APIRouter:
    """
    Return kwik's built-in routes as one router, minus the excluded ``(METHOD, path)`` pairs.

    Apps that override a built-in route register their own handler first and exclude
    kwik's, so the application has one handler per pair and OpenAPI documents the one
    that runs. Route objects move across untouched, so the routes that stay keep their
    response models, dependencies and OpenAPI metadata.

    Raises:
        ValueError: if a pair matches no route (a kwik upgrade moved it: fail at import
            instead of silently restoring it), or matches a route that also serves
            methods that are not excluded (dropping it would withdraw them too).

    """
    excluded = {(method.upper(), path) for method, path in exclude}
    built = APIRouter()
    matched: set[tuple[str, str]] = set()
    for router in (login_router, users_router, roles_router, permissions_router):
        for route in router.routes:
            path = getattr(route, "path", None)
            pairs = {(method, path) for method in getattr(route, "methods", None) or ()}
            hits = pairs & excluded
            if not hits:
                built.routes.append(route)
                continue
            if hits != pairs:
                surviving = sorted(method for method, _ in pairs - hits)
                msg = f"kwik route {path} also serves {surviving}; excluding {sorted(m for m, _ in hits)} would withdraw them"
                raise ValueError(msg)
            matched |= hits
    missing = sorted(excluded - matched)
    if missing:
        msg = f"kwik has no built-in route for {missing}"
        raise ValueError(msg)
    return built


api_router = build_api_router()
```
